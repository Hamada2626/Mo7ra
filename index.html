function elapsed(date) { if (!date) return 'اختار التاريخ ❤️'; const start = new Date(date); const now = new Date(); if (start > now) return 'التاريخ لسه مجاش ❤️'; let years = now.getFullYear() - start.getFullYear(); let months = now.getMonth() - start.getMonth(); let days = now.getDate() - start.getDate(); if (days < 0) { months--; const previousMonth = new Date( now.getFullYear(), now.getMonth(), 0 ); days += previousMonth.getDate(); } if (months < 0) { years--; months += 12; } let seconds = Math.floor((now - start) / 1000); let hours = Math.floor(seconds / 3600) % 24; let minutes = Math.floor(seconds / 60) % 60; seconds %= 60; return ` ${years} سنة ❤️
${months} شهر
${days} يوم
${hours} ساعة ${minutes} دقيقة ${seconds} ثانية `; }
